            close(conn_fd);

            if (thread_error != 0) {
                server_log_errno("pthread_join fallback", thread_error);
                _exit(1);
            }
        }
    }

    // detached 스레드가 설정과 라우트를 다 쓸 때까지 main이 정리하지 않습니다.
    int wait_error = pthread_mutex_lock(&g_active_mutex);
    if (wait_error != 0) {
        server_log_errno("pthread_mutex_lock shutdown", wait_error);
        _exit(1);
    }

        wait_error = pthread_cond_wait(&g_active_changed, &g_active_mutex);
        if (wait_error != 0) {
            pthread_mutex_unlock(&g_active_mutex);
            server_log_errno("pthread_cond_wait shutdown", wait_error);
            _exit(1);
        }
    }
    wait_error = pthread_mutex_unlock(&g_active_mutex);
    if (wait_error != 0) {
        server_log_errno("pthread_mutex_unlock shutdown", wait_error);
        _exit(1);
    }
    pthread_cond_destroy(&g_active_changed);
    pthread_mutex_destroy(&g_active_mutex);
    server_log("server stopped\n");
    cleanup_routes(routes, cfg.route_count);
    return 0;
}

        size_t end = pos;
        while (end < length && value[end] != ',') end++;
        size_t left = pos, right = end;

        while (right > left && (value[right - 1] == ' ' ||

        size_t equals = left;

        size_t name_end = equals;

                                   value[name_end - 1] == '\t')) name_end--;

            span_equals(value + left, name_end - left, "no-store") ||
            span_equals(value + left, name_end - left, "no-cache") ||

static int header_end(const unsigned char *data, size_t length, size_t *end) {

static int forbidden_request_header(const struct http_request *req) {
    static const char *const names[] = {
        "Authorization", "Cookie", "Range", "If-Modified-Since",
        "If-None-Match", "Cache-Control", "Pragma"
    };
    for (int i = 0; i < req->nheaders; i++) {
        for (size_t j = 0; j < sizeof(names) / sizeof(names[0]); j++) {
            if (strcasecmp(req->headers[i].name, names[j]) == 0) return 1;
        }
    }
    return 0;
}

int cache_key_for_request(const struct http_request *req,
                          char *key, size_t capacity) {
    if (req == NULL || key == NULL || capacity == 0 ||
        strcmp(req->method, "GET") != 0 || forbidden_request_header(req)) {
        return 0;
    }

    return 1;
}

static int transformed_header(const unsigned char *raw, size_t raw_length,
                              const char *cache_status, int keep_alive,
                              unsigned char **out, size_t *out_length,
                              int *status, size_t *content_length,
                              int *has_content_length, int *safe_to_store,
                              size_t *ttl_seconds) {
    size_t end;
    if (!header_end(raw, raw_length, &end)) return 0;
    size_t first_line = 0;
    while (first_line + 1 < end &&
           !(raw[first_line] == '\r' && raw[first_line + 1] == '\n')) {
        first_line++;
    }
    if (first_line + 1 >= end || first_line < 12 ||
        memcmp(raw, "HTTP/", 5) != 0) return -1;
    *status = 0;
    for (size_t i = 9; i < first_line && raw[i] >= '0' && raw[i] <= '9'; i++) {
        *status = *status * 10 + (raw[i] - '0');
    }
    *content_length = 0;
    *has_content_length = 0;
    *safe_to_store = 1;
    *ttl_seconds = CACHE_TTL_SECONDS;
    size_t capacity = end + 128;
    unsigned char *built = malloc(capacity);
    if (built == NULL) return -1;
    size_t used = 0;
    memcpy(built, raw, first_line + 2);
    used = first_line + 2;

    size_t pos = used;
    while (pos + 1 < end) {
        size_t line_end = pos;
        while (line_end + 1 < end &&
               !(raw[line_end] == '\r' && raw[line_end + 1] == '\n')) {
            line_end++;
        }
        if (line_end == pos) break;
        size_t colon = pos;
        while (colon < line_end && raw[colon] != ':') colon++;
        size_t name_len = colon - pos;
        if (colon < line_end && name_len == 14 &&
            strncasecmp((const char *)raw + pos, "Content-Length", name_len) == 0) {
            char number[32];
            size_t value_start = colon + 1;
            while (value_start < line_end && raw[value_start] == ' ') value_start++;
            size_t value_len = line_end - value_start;
            if (value_len == 0 || value_len >= sizeof(number)) *safe_to_store = 0;
            else {
                memcpy(number, raw + value_start, value_len);
                number[value_len] = '\0';
                char *tail = NULL;
                errno = 0;
                unsigned long long parsed = strtoull(number, &tail, 10);
                if (errno != 0 || tail == number || *tail != '\0' ||
                    parsed > CACHE_ITEM_MAX) *safe_to_store = 0;
                else {
                    *content_length = (size_t)parsed;
                    *has_content_length = 1;
                }
            }
        }
        if (colon < line_end &&
            ((name_len == 10 && strncasecmp((const char *)raw + pos,
                                            "Connection", 10) == 0) ||
             (name_len == 7 && strncasecmp((const char *)raw + pos,
                                           "X-Cache", 7) == 0))) {
            // Connection and X-Cache are regenerated for each response.
        } else {
            memcpy(built + used, raw + pos, line_end + 2 - pos);
            used += line_end + 2 - pos;
        }
        if (colon < line_end &&
            ((name_len == 17 && strncasecmp((const char *)raw + pos,
                                            "Transfer-Encoding", 17) == 0) ||
             (name_len == 4 && strncasecmp((const char *)raw + pos,
                                           "Vary", 4) == 0) ||
             (name_len == 10 && strncasecmp((const char *)raw + pos,
                                            "Set-Cookie", 10) == 0) ||
             (name_len == 7 && strncasecmp((const char *)raw + pos,
                                           "Expires", 7) == 0))) {
            *safe_to_store = 0;
        }
        if (colon < line_end && name_len == 13 &&
            strncasecmp((const char *)raw + pos, "Cache-Control", 13) == 0) {
            size_t value_len = line_end - colon - 1;
            apply_cache_control(raw + colon + 1, value_len,
                                ttl_seconds, safe_to_store);
        }
        pos = line_end + 2;
    }
    // 본문 길이를 알 때만 연결을 유지해야 다음 요청과의 경계가 분명합니다.
    int keep = keep_alive && *has_content_length;
    int added = snprintf((char *)built + used, capacity - used,
                         "X-Cache: %s\r\nConnection: %s\r\n\r\n",
                         cache_status, keep ? "keep-alive" : "close");
    if (added < 0 || (size_t)added >= capacity - used) {
        free(built);
        return -1;
    }
    used += (size_t)added;
    *out = built;
    *out_length = used;
    return 1;
}

    (void)has_length;
    (void)safe;

static int append_bytes(unsigned char **buffer, size_t *used, size_t *capacity,
                        const void *data, size_t length, size_t limit) {
    if (length > limit - *used) return -1;
    size_t needed = *used + length;
    if (needed > *capacity) {
        size_t next = *capacity == 0 ? 1024 : *capacity;
        while (next < needed) next = next > limit / 2 ? limit : next * 2;
        unsigned char *grown = realloc(*buffer, next);
        if (grown == NULL) return -1;
        *buffer = grown;
        *capacity = next;
    }
    memcpy(*buffer + *used, data, length);
    *used = needed;
    return 0;
}

int cache_capture_finish(struct cache_capture *capture) {
    if (capture == NULL) return -1;
    if (!capture->header_sent && capture->pending_len > 0) {
        // 헤더가 완성되지 않은 응답은 MISS 표시 없이 그대로 전달합니다.
        if (net_send_all(capture->client_fd, capture->pending,
                         capture->pending_len) < 0) {
            cache_capture_abort(capture);
            return -1;
        }
    }
    if (capture->cacheable_response && capture->expected_length == capture->body_length &&
        capture->response_len <= CACHE_ITEM_MAX) {
        struct timespec now;
        struct cache_entry *entry = calloc(1, sizeof(*entry));
        unsigned char *saved = malloc(capture->response_len);
        if (entry != NULL && saved != NULL &&
            clock_gettime(CLOCK_MONOTONIC, &now) == 0) {
            memcpy(entry->key, capture->key, strlen(capture->key) + 1);
            entry->has_file_source = capture->has_file_source;
            if (capture->has_file_source) {
                memcpy(entry->file_source, capture->file_source,
                       strlen(capture->file_source) + 1);
                entry->file_source_stat = capture->file_source_stat;
            }
            memcpy(saved, capture->response, capture->response_len);
            entry->response = saved;
            saved = NULL;
            entry->length = capture->response_len;
            entry->expires_at = now;
            entry->expires_at.tv_sec += (time_t)capture->ttl_seconds;
            pthread_mutex_lock(&cache_mutex);
            remove_expired_locked(&now);
            if (capture->generation == cache_generation) {
                struct cache_entry **link = &cache_head;
                while (*link != NULL) {
                    if (strcmp((*link)->key, entry->key) == 0) {
                        struct cache_entry *old = *link;
                        *link = old->next;
                        if (cache_tail == old) cache_tail = NULL;
                        cache_entries--;
                        cache_bytes -= old->length;
                        free(old->response);
                        free(old);
                        break;
                    }
                    link = &(*link)->next;
                }
                if (cache_tail == NULL && cache_head != NULL) {
                    cache_tail = cache_head;
                    while (cache_tail->next != NULL) cache_tail = cache_tail->next;
                }
                while (cache_head != NULL &&
                       (cache_entries >= CACHE_ENTRY_LIMIT ||
                        cache_bytes + entry->length > CACHE_TOTAL_LIMIT)) {
                    remove_oldest_locked();
                }
                if (entry->length <= CACHE_TOTAL_LIMIT) {
                    if (cache_tail == NULL) cache_head = cache_tail = entry;
                    else { cache_tail->next = entry; cache_tail = entry; }
                    cache_entries++;
                    cache_bytes += entry->length;
                    entry = NULL;
                }
            }
            pthread_mutex_unlock(&cache_mutex);
        }
        if (entry != NULL) {
            free(entry->response);
            free(entry);
        }
        free(saved);
    }
    cache_capture_abort(capture);
    return 0;
}

void cache_capture_abort(struct cache_capture *capture) {
    if (capture == NULL) return;
    free(capture->pending);
    free(capture->response);
    memset(capture, 0, sizeof(*capture));
}

void cache_invalidate_all(void) {
    pthread_mutex_lock(&cache_mutex);
    while (cache_head != NULL) remove_oldest_locked();
    // 진행 중인 이전 GET 응답이 변경 뒤에 낡은 항목을 저장하지 못하게 합니다.
    cache_generation++;
    pthread_mutex_unlock(&cache_mutex);
}

void cache_destroy(void) {
    pthread_mutex_lock(&cache_mutex);
    while (cache_head != NULL) remove_oldest_locked();
    pthread_mutex_unlock(&cache_mutex);
}

        return cgi_error(fd, 500);
    }
    for (size_t i = 0; i < sizeof(env_results) / sizeof(env_results[0]); i++) {
        if (env_results[i] < 0 || (size_t)env_results[i] >= env_capacities[i]) {
            return cgi_error(fd, 500);
        }
    }
    char *envp[] = {env_method, env_query, env_length, env_type, env_script,
                    env_protocol, "GATEWAY_INTERFACE=CGI/1.1",
                    "PATH=/usr/bin:/bin", NULL};

    if (pipe(input_pipe) < 0) return cgi_error(fd, 500);
    if (pipe(output_pipe) < 0) {
        close(input_pipe[0]); close(input_pipe[1]);
        return cgi_error(fd, 500);
    }

    pid_t pid = fork();
    if (pid < 0) {
        close(input_pipe[0]); close(input_pipe[1]);
        close(output_pipe[0]); close(output_pipe[1]);
        return cgi_error(fd, 500);
    }
    if (pid == 0) {
        if (dup2(input_pipe[0], STDIN_FILENO) < 0 ||
            dup2(output_pipe[1], STDOUT_FILENO) < 0) _exit(127);
        close(input_pipe[0]); close(input_pipe[1]);
        close(output_pipe[0]); close(output_pipe[1]);

        char *argv[] = {script, NULL};
        execve(script, argv, envp);
        _exit(127);
    }

    close(input_pipe[0]);
    close(output_pipe[1]);
    if (body_len > 0 && write_all(input_pipe[1], body, body_len) < 0) {
        close(input_pipe[1]); close(output_pipe[0]);
        kill(pid, SIGKILL);
        waitpid(pid, NULL, 0);
        return cgi_error(fd, 500);
    }
    close(input_pipe[1]);

    char *output = malloc(CGI_OUTPUT_LIMIT + 1);
    if (output == NULL) {
        close(output_pipe[0]);
        kill(pid, SIGKILL);
        waitpid(pid, NULL, 0);
        return cgi_error(fd, 500);
    }
    size_t output_len = 0;
    int timed_out = 0;
    int too_large = 0;
    for (;;) {
        int ready = net_wait(output_pipe[0], POLLIN, cfg->timeout_sec * 1000);
        if (ready == 0) { timed_out = 1; break; }
        if (ready < 0 && errno == EINTR) continue;
        if (ready < 0) break;
        ssize_t got = read(output_pipe[0], output + output_len,
                           CGI_OUTPUT_LIMIT - output_len);
        if (got < 0 && errno == EINTR) continue;
        if (got < 0) break;
        if (got == 0) break;
        output_len += (size_t)got;
        if (output_len == CGI_OUTPUT_LIMIT) {
            char extra;
            ssize_t more = read(output_pipe[0], &extra, 1);
            if (more > 0) too_large = 1;
            break;
        }
    }
    close(output_pipe[0]);
    if (timed_out || too_large) kill(pid, SIGKILL);
    int child_status = 0;
    while (waitpid(pid, &child_status, 0) < 0) {
        if (errno != EINTR) { free(output); return cgi_error(fd, 500); }
    }
    if (timed_out) { free(output); return cgi_error(fd, 504); }
    if (too_large || !WIFEXITED(child_status) || WEXITSTATUS(child_status) != 0) {
        free(output);
        return cgi_error(fd, 500);
    }

    size_t header_end = 0;
    if (!find_header_end(output, output_len, &header_end)) {
        free(output);
        return cgi_error(fd, 500);
    }
    char content_type[256] = "text/plain; charset=utf-8";
    size_t line = 0;
    while (line + 1 < header_end) {
        size_t end = line;
        while (end + 1 < header_end &&
               !(output[end] == '\r' && output[end + 1] == '\n')) end++;
        if (end == line) break;
        const char *colon = memchr(output + line, ':', end - line);
        if (colon != NULL) {
            size_t name_len = (size_t)(colon - (output + line));
            if (name_len == 12 && strncasecmp(output + line, "Content-Type", 12) == 0) {
                size_t value = (size_t)(colon - output) + 1;
                while (value < end && (output[value] == ' ' || output[value] == '\t')) value++;
                size_t value_len = end - value;
                if (value_len >= sizeof(content_type)) value_len = sizeof(content_type) - 1;
                memcpy(content_type, output + value, value_len);
                content_type[value_len] = '\0';
            }
        }
        line = end + 2;
    }
    size_t body_start = header_end;
    int send_result = resp_send_simple(fd, 200, content_type,
                                       output + body_start, output_len - body_start,
                                       req->keep_alive);
    free(output);
    return send_result < 0 ? -1 : 0;
}

static int parse_number(const char *text, int min, int max, int *out) {
    if (text == NULL || *text == '\0') {
        return -1;
    }
    for (const char *p = text; *p != '\0'; p++) {
        if (*p < '0' || *p > '9') {
            return -1;
        }
    }
    errno = 0;
    char *end = NULL;
    long value = strtol(text, &end, 10);
    if (errno == ERANGE || end == text || *end != '\0' ||
        value < min || value > max || value > INT_MAX) {
        return -1;
    }
    *out = (int)value;
    return 0;
}

// 이 값들은 TCP 상태가 아니라 서버 프로그램이 관리하는 처리 단계입니다.
enum conn_state {
    CONN_READING,
    CONN_PARSING,
    CONN_ROUTING,
    CONN_RESPONDING,
    CONN_KEEPALIVE_CHECK,
    CONN_CLOSING,
    CONN_CLOSED
};

// 연결 하나를 처리하는 동안 필요한 번호, 현재 단계, 요청 저장 공간입니다.
struct conn {
    int fd;
    enum conn_state state;
    char rbuf[8192];
    size_t rlen;
    size_t request_len;
    int keep_alive;
    int response_sent;
    char *body;
    size_t body_len;
};

// 상태 출력에 사용할 단계 이름을 돌려줍니다.
static const char *conn_state_name(enum conn_state state) {
    switch (state) {
        case CONN_READING: return "READING";
        case CONN_PARSING: return "PARSING";
        case CONN_ROUTING: return "ROUTING";
        case CONN_RESPONDING: return "RESPONDING";
        case CONN_KEEPALIVE_CHECK: return "KEEPALIVE_CHECK";
        case CONN_CLOSING: return "CLOSING";
        case CONN_CLOSED: return "CLOSED";
    }
    return "UNKNOWN";
}

// 단계가 바뀌면 서버 터미널의 오류 출력(stderr)에 전환을 기록합니다.
static void conn_set_state(struct conn *c, enum conn_state next) {
    server_log("fd %d: %s -> %s\n", c->fd,
               conn_state_name(c->state), conn_state_name(next));
    c->state = next;
}

// 요청 버퍼 안에서 CRLF 두 쌍을 길이 제한 안에 찾아냅니다.
static int conn_has_header_end(const char *buffer, size_t length) {
    if (length < 4) return 0;
    for (size_t i = 0; i <= length - 4; i++) {
        if (memcmp(buffer + i, "\r\n\r\n", 4) == 0) return 1;
    }
    return 0;
}

// 파싱 실패 결과에 맞는 짧은 HTTP 오류 응답을 고릅니다.
static int conn_error_status(enum http_result result) {
    switch (result) {
        case HTTP_BAD_REQUEST: return 400;
        case HTTP_URI_TOO_LONG: return 414;
        case HTTP_HEADER_TOO_LARGE: return 431;
        case HTTP_NOT_IMPLEMENTED: return 501;
        case HTTP_VERSION_NOT_SUPPORTED: return 505;
        case HTTP_PAYLOAD_TOO_LARGE: return 413;
        case HTTP_OK:
        case HTTP_INCOMPLETE:
            return 500;
    }
    return 500;
}

void conn_handle(int fd, const char *client_ip, const struct config *cfg) {
    struct conn c = {
        .fd = fd,
        .state = CONN_READING,
        .rlen = 0
    };

    // 현재 단계가 닫힘에 도달할 때까지 이 연결의 처리 단계를 진행합니다.
    while (c.state != CONN_CLOSED) {
        switch (c.state) {
            case CONN_READING: {
                // 앞 요청을 처리하고도 다음 요청 바이트가 남았는지 먼저 확인합니다.
                if (conn_has_header_end(c.rbuf, c.rlen)) {
                    server_log("fd %d: request data available (%zu bytes)\n",
                               c.fd, c.rlen);
                    conn_set_state(&c, CONN_PARSING);
                    break;
                }

                // 아직 비어 있는 버퍼 영역의 크기를 구해 넘칠 공간이 없도록 합니다.
                size_t available = sizeof(c.rbuf) - c.rlen - 1;
                if (available == 0) {
                    server_log("fd %d: request headers exceed buffer capacity\n",
                               c.fd);
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }

                // 커널에 새 요청 바이트를 기존 데이터 뒤에 이어서 rbuf에 복사해 달라고 요청합니다.
                ssize_t n = recv(c.fd, c.rbuf + c.rlen, available, 0);
                if (n < 0) {
                    if (errno == EINTR) break;
                    if (errno == EAGAIN || errno == EWOULDBLOCK) {
                        if (c.rlen > 0) {
                            resp_send_error(c.fd, 408);
                        }
                        server_log("fd %d: request read timed out\n", c.fd);
                    } else {
                        server_log_errno("recv", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                } else if (n == 0) {
                    server_log("fd %d: peer closed\n", c.fd);
                    conn_set_state(&c, CONN_CLOSING);
                } else {
                    // 이번에 읽은 바이트 수를 누적하고 문자열 끝을 표시합니다.
                    c.rlen += (size_t)n;
                    c.rbuf[c.rlen] = '\0';

                    // HTTP 헤더 끝의 빈 줄이 도착했을 때만 요청 읽기를 마칩니다.
                    if (conn_has_header_end(c.rbuf, c.rlen)) {
                        server_log("fd %d: request data available (%zu bytes)\n",
                                   c.fd, c.rlen);
                        conn_set_state(&c, CONN_PARSING);
                    } else if (c.rlen == sizeof(c.rbuf) - 1) {
                        // 용량을 다 썼는데 끝 표시가 없으면 버퍼를 더 쓰지 않고 연결을 닫습니다.
                        server_log("fd %d: request headers exceed buffer capacity\n",
                                   c.fd);
                        conn_set_state(&c, CONN_CLOSING);
                    }
                }
                break;
            }

            case CONN_PARSING: {
                // 원문 요청 줄을 검증하고 method, target, version 필드로 나눕니다.
                struct http_request req;
                enum http_result result = http_parse(c.rbuf, c.rlen, &req);
                server_log("fd %d: parse result %s\n", c.fd,
                           http_result_name(result));
                if (result == HTTP_OK) {
                    c.request_len = req.header_len + (size_t)req.content_length;
                    const char *connection = http_header_get(&req, "Connection");
                    c.keep_alive = strcmp(req.version, "HTTP/1.1") == 0;
                    if (connection != NULL) {
                        if (strcasecmp(connection, "close") == 0) {
                            c.keep_alive = 0;
                        } else if (strcasecmp(connection, "keep-alive") == 0) {
                            c.keep_alive = 1;
                        }
                    }
                    req.keep_alive = c.keep_alive;
                    server_log("fd %d: request %s %s %s; headers=%d bytes=%zu\n",
                               c.fd, req.method, req.target, req.version,
                               req.nheaders, req.header_len);
                    if (req.content_length > 0) {
                        size_t expected = (size_t)req.content_length;
                        c.body = malloc(expected + 1);
                        if (c.body == NULL) {
                            server_log_errno("malloc request body", errno);
                            conn_set_state(&c, CONN_CLOSING);
                            break;
                        }

                        // 헤더와 함께 이미 받은 본문 바이트를 새 저장 공간으로 옮깁니다.
                        size_t received = c.rlen - req.header_len;
                        if (received > expected) {
                            received = expected;
                        }
                        memcpy(c.body, c.rbuf + req.header_len, received);

                        // Content-Length가 가리키는 나머지 바이트만 더 받습니다.
                        while (received < expected) {
                            ssize_t n = recv(c.fd, c.body + received,
                                             expected - received, 0);
                            if (n < 0) {
                                if (errno == EINTR) continue;
                                if (errno == EAGAIN || errno == EWOULDBLOCK) {
                                    resp_send_error(c.fd, 408);
                                } else {
                                    server_log_errno("recv body", errno);
                                }
                                free(c.body);
                                c.body = NULL;
                                conn_set_state(&c, CONN_CLOSING);
                                break;
                            }
                            if (n == 0) {
                                server_log("fd %d: peer closed during request body\n",
                                           c.fd);
                                free(c.body);
                                c.body = NULL;
                                conn_set_state(&c, CONN_CLOSING);
                                break;
                            }
                            received += (size_t)n;
                        }
                        if (c.state == CONN_CLOSING) {
                            break;
                        }
                        c.body[received] = '\0';
                        c.body_len = received;
                        server_log("fd %d: request body bytes=%zu\n",
                                   c.fd, received);
                    }
                    conn_set_state(&c, CONN_ROUTING);
                } else {
                    if (resp_send_error(c.fd, conn_error_status(result)) < 0) {
                        server_log_errno("send error response", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                }
                break;
            }

            case CONN_ROUTING: {
                // 프록시 규칙이 먼저 맞는지 보고, 아니면 정적 파일이나 CGI로 보냅니다.
                struct http_request req;
                enum http_result parse_result = http_parse(c.rbuf, c.rlen, &req);
                if (parse_result != HTTP_OK) {
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                req.keep_alive = c.keep_alive;
                int serve_result = 0;
                int proxy_handled = 0;
                int invalidates_cache = strcmp(req.method, "GET") != 0;
                if (cache_request_lock(invalidates_cache) < 0) {
                    server_log_errno("cache request lock", errno);
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                if (invalidates_cache) {
                    // 쓰기 잠금 안에서 기존 응답을 지워 이후 읽기가 원본을 확인하게 합니다.
                    cache_invalidate_all();
                }
                struct cache_capture capture;
                struct cache_capture *capture_ptr = NULL;
                char cache_key[CACHE_KEY_MAX];
                if (strncmp(req.target, "/cgi-bin/", 9) != 0 &&
                    cache_key_for_request(&req, cache_key, sizeof(cache_key))) {
                    int cache_result = cache_lookup_send(c.fd, cache_key,
                                                         c.keep_alive);
                    if (cache_result < 0) {
                        cache_request_unlock();
                        conn_set_state(&c, CONN_CLOSING);
                        break;
                    }
                    if (cache_result > 0) {
                        // HIT 응답은 저장된 Content-Length로 끝나므로 연결을 유지할 수 있습니다.
                        c.response_sent = 1;
                        cache_request_unlock();
                        conn_set_state(&c, CONN_RESPONDING);
                        break;
                    }
                    if (cache_capture_begin(&capture, c.fd, cache_key) == 0) {
                        capture_ptr = &capture;
                    }
                }
                if (cfg->forward_proxy &&
                    (strncmp(req.target, "http://", 7) == 0 ||
                     strncmp(req.target, "https://", 8) == 0)) {
                    serve_result = proxy_forward_cached(c.fd, &req, c.body,
                                                        c.body_len,
                                                        cfg->timeout_sec,
                                                        client_ip, capture_ptr);
                    proxy_handled = 1;
                } else {
                    int reverse_result = proxy_reverse_cached(c.fd, &req, c.body,
                                                              c.body_len, cfg,
                                                              client_ip,
                                                              capture_ptr);
                    if (reverse_result == PROXY_NO_ROUTE) {
                        if (strncmp(req.target, "/cgi-bin/", 9) == 0) {
                            serve_result = cgi_serve(c.fd, cfg, &req,
                                                     c.body, c.body_len);
                        } else {
                            serve_result = static_serve_cached(c.fd, cfg, &req,
                                                               capture_ptr);
                        }
                    } else {
                        serve_result = reverse_result == PROXY_FAILED ? -1 : 0;
                        proxy_handled = 1;
                    }
                }
                if (invalidates_cache) {
                    // 변경 응답을 마친 뒤 다시 비워 변경 중 생성된 항목도 제거합니다.
                    cache_invalidate_all();
                }
                if (serve_result < 0) {
                    cache_capture_abort(capture_ptr);
                    cache_request_unlock();
                    if (errno != EPIPE && errno != ECONNRESET) {
                        server_log_errno("static_serve", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                if (capture_ptr != NULL && !capture_ptr->keep_alive)
                    c.keep_alive = 0;
                if (capture_ptr != NULL && cache_capture_finish(capture_ptr) < 0) {
                    cache_request_unlock();
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                if (cache_request_unlock() < 0) {
                    server_log_errno("cache request unlock", errno);
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                c.response_sent = 1;
                // 오류 응답과 프록시 중계(업스트림 EOF까지 전달)는 연결을 닫습니다.
                // 캐시 MISS 정적 응답은 cache_capture가 keep-alive 헤더를 맞춰 보냅니다.
                if (serve_result > 0 || proxy_handled)
                    c.keep_alive = 0;
                conn_set_state(&c, CONN_RESPONDING);
                break;
            }

            case CONN_RESPONDING: {
                if (c.response_sent) {
                    conn_set_state(&c, CONN_KEEPALIVE_CHECK);
                    break;
                }
                static const char body[] = "hello\n";
                if (resp_send_simple(c.fd, 200, "text/plain; charset=utf-8",
                                     body, sizeof(body) - 1, c.keep_alive) < 0) {
                    if (errno != EPIPE && errno != ECONNRESET) {
                        server_log_errno("send", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                conn_set_state(&c, CONN_KEEPALIVE_CHECK);
                break;
            }

            case CONN_KEEPALIVE_CHECK:
                if (!c.keep_alive) {
                    conn_set_state(&c, CONN_CLOSING);
                    break;

        ssize_t n = send(fd, bytes + sent, len - sent, 0);

static void *pool_worker(void *argument) {
    struct pool *p = argument;

    for (;;) {
        pthread_mutex_lock(&p->mutex);
        while (p->count == 0 && !p->stopping) {
            pthread_cond_wait(&p->not_empty, &p->mutex);
        }

        if (p->count == 0 && p->stopping) {
            pthread_mutex_unlock(&p->mutex);
            return NULL;
        }

        int fd = p->queue[p->head];
        p->head = (p->head + 1) % p->queue_cap;
        p->count--;
        pthread_cond_signal(&p->not_full);
        pthread_mutex_unlock(&p->mutex);

        p->handler(fd);
    }
}

    p->not_full_ready = 1;

    for (int i = 0; i < workers; i++) {
        result = pthread_create(&p->threads[i], NULL, pool_worker, p);
        if (result != 0) {
            errno = result;
            goto fail;
        }
        p->started_count++;
    }
    return p;

fail:
    pool_shutdown(p);
    return NULL;
}

// 큐가 가득 찬 동안 기다렸다가 연결 fd를 원형 대기열에 넣습니다.
int pool_submit(struct pool *p, int fd) {
    if (p == NULL) {
        errno = EINVAL;
        return -1;
    }

    int result = pthread_mutex_lock(&p->mutex);
    if (result != 0) {
        errno = result;
        return -1;
    }

    while (p->count == p->queue_cap && !p->stopping) {
        result = pthread_cond_wait(&p->not_full, &p->mutex);
        if (result != 0) {
            pthread_mutex_unlock(&p->mutex);
            errno = result;
            return -1;
        }
    }

    if (p->stopping) {
        pthread_mutex_unlock(&p->mutex);
        errno = ECANCELED;
        return -1;
    }

    p->queue[p->tail] = fd;
    p->tail = (p->tail + 1) % p->queue_cap;
    p->count++;
    pthread_cond_signal(&p->not_empty);
    pthread_mutex_unlock(&p->mutex);
    return 0;
}

// 제출을 막고 대기 중인 일꾼을 깨운 뒤 큐를 비우고 자원을 해제합니다.
void pool_shutdown(struct pool *p) {
    if (p == NULL) return;

    if (p->mutex_ready) {
        pthread_mutex_lock(&p->mutex);
        p->stopping = 1;
        if (p->not_empty_ready) pthread_cond_broadcast(&p->not_empty);
        if (p->not_full_ready) pthread_cond_broadcast(&p->not_full);
        pthread_mutex_unlock(&p->mutex);
    }

    for (int i = 0; i < p->started_count; i++) {
        pthread_join(p->threads[i], NULL);
    }

    if (p->not_full_ready) pthread_cond_destroy(&p->not_full);
    if (p->not_empty_ready) pthread_cond_destroy(&p->not_empty);
    if (p->mutex_ready) pthread_mutex_destroy(&p->mutex);
    free(p->queue);
    free(p->threads);
    free(p);
}

                         const char *format, ...) {

static char *build_request(const struct http_request *req,
                           const char *request_target,
                           const char *host_header,
                           size_t body_len, const char *client_ip,
                           size_t *header_len) {
    char *buffer = malloc(PROXY_HEADER_CAP);
    if (buffer == NULL) return NULL;
    size_t used = 0;

    if (append_format(buffer, PROXY_HEADER_CAP, &used,
                      "%s %s HTTP/1.1\r\n", req->method,
                      request_target) < 0 ||
        append_format(buffer, PROXY_HEADER_CAP, &used,
                      "Host: %s\r\n", host_header) < 0) {
        free(buffer);
        errno = EOVERFLOW;
        return NULL;
    }

    int had_content_length = 0;
    for (int i = 0; i < req->nheaders; i++) {
        const struct http_header *header = &req->headers[i];
        if (!valid_header_name(header->name) ||
            !valid_header_value(header->value)) {
            free(buffer);
            errno = EINVAL;
            return NULL;
        }
        if (strcasecmp(header->name, "Content-Length") == 0) {
            had_content_length = 1;
            continue;
        }
        if (strcasecmp(header->name, "Host") == 0 ||
            strcasecmp(header->name, "X-Forwarded-For") == 0 ||
            is_hop_by_hop(req, header->name)) {
            continue;
        }
        if (append_format(buffer, PROXY_HEADER_CAP, &used, "%s: %s\r\n",
                          header->name, header->value) < 0) {
            free(buffer);
            errno = EOVERFLOW;
            return NULL;
        }
    }

    if (had_content_length || body_len > 0) {
        if (append_format(buffer, PROXY_HEADER_CAP, &used,
                          "Content-Length: %zu\r\n", body_len) < 0) {
            free(buffer);
            errno = EOVERFLOW;
            return NULL;
        }
    }
    if (append_format(buffer, PROXY_HEADER_CAP, &used,
                      "X-Forwarded-For: %s\r\nConnection: close\r\n\r\n",
                      client_ip != NULL ? client_ip : "unknown") < 0) {
        free(buffer);
        errno = EOVERFLOW;
        return NULL;
    }

    *header_len = used;
    return buffer;
}

// 연결·요청 전송·응답 중계를 한곳에서 처리하고 만든 소켓을 항상 닫습니다.
static int proxy_exchange_connected(int client_fd,
                                    const struct http_request *req,
                                    const char *body, size_t body_len,
                                    int upstream_fd, const char *host_header,
                                    const char *request_target,
                                    const char *client_ip,
                                    int timeout_ms,
                                    struct cache_capture *capture) {
    size_t request_len = 0;
    char *request = build_request(req, request_target, host_header,
                                  body_len, client_ip, &request_len);
    if (request == NULL) {
        int saved_error = errno;
        server_log_errno("build proxy request", saved_error);
        close(upstream_fd);
        return resp_send_error(client_fd, 500) < 0 ? PROXY_FAILED : PROXY_HANDLED;
    }

    if (net_send_all(upstream_fd, request, request_len) < 0 ||
        (body_len > 0 && net_send_all(upstream_fd, body, body_len) < 0)) {
        int saved_error = errno;
        free(request);
        close(upstream_fd);
        server_log("upstream request send failed (errno=%d)\n", saved_error);
        return resp_send_error(client_fd, 502) < 0 ? PROXY_FAILED : PROXY_HANDLED;
    }
    free(request);

    char buffer[PROXY_IO_CAP];
    size_t relayed = 0;
    int result = PROXY_HANDLED;
    for (;;) {
        int ready;
        do {
            ready = net_wait(upstream_fd, POLLIN, timeout_ms);
        } while (ready < 0 && errno == EINTR);

        if (ready == 0) {
            if (relayed == 0) {
                result = resp_send_error(client_fd, 504) < 0 ?
                         PROXY_FAILED : PROXY_HANDLED;
            }
            break;
        }
        if (ready < 0) {
            int saved_error = errno;
            if (relayed == 0) {
                server_log("upstream wait failed (errno=%d)\n", saved_error);
                result = resp_send_error(client_fd, 502) < 0 ?
                         PROXY_FAILED : PROXY_HANDLED;
            }
            break;
        }

        ssize_t count = recv(upstream_fd, buffer, sizeof(buffer), 0);
        if (count < 0 && errno == EINTR) continue;
        if (count < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
            if (relayed == 0) {
                result = resp_send_error(client_fd, 504) < 0 ?
                         PROXY_FAILED : PROXY_HANDLED;
            }
            break;
        }
        if (count < 0) {
            int saved_error = errno;
            if (relayed == 0) {
                server_log("upstream receive failed (errno=%d)\n", saved_error);
                result = resp_send_error(client_fd, 502) < 0 ?
                         PROXY_FAILED : PROXY_HANDLED;
            }
            break;
        }
        if (count == 0) break;

        int send_result = capture != NULL ?
            cache_capture_write(capture, buffer, (size_t)count) :
            net_send_all(client_fd, buffer, (size_t)count);
        if (send_result < 0) {
            int saved_error = errno;
            if (saved_error != EPIPE && saved_error != ECONNRESET) {
                server_log("client response send failed (errno=%d)\n",
                           saved_error);
            }
            result = PROXY_FAILED;
            break;
        }
        relayed += (size_t)count;
    }

    close(upstream_fd);
    return result;
}

// 업스트림에 연결하고, 연결 실패를 502 또는 504 응답으로 돌려줍니다.
static int proxy_exchange(int client_fd, const struct http_request *req,

                          const struct upstream *upstream,
                          const char *request_target, const char *client_ip,

    if (upstream_fd < 0) {

                   saved_error, status);
        return resp_send_error(client_fd, status) < 0 ? PROXY_FAILED : PROXY_HANDLED;

    if (route == NULL) return PROXY_NO_ROUTE;

    int timeout_ms = cfg->timeout_sec * 1000;
    int last_error = ECONNREFUSED;

        memset(&upstream, 0, sizeof(upstream));
        memcpy(upstream.host, backend->host, strlen(backend->host) + 1);

}

// http:// authority와 경로를 나눠 포워드 프록시 목적지를 구성합니다.
static int parse_absolute_target(const char *target, struct upstream *upstream,
                                  char *origin, size_t origin_capacity) {
    if (strncmp(target, "https://", 8) == 0) return 400;
    if (strncmp(target, "http://", 7) != 0) return 400;

    const char *authority = target + 7;
    const char *path = strpbrk(authority, "/?#");
    size_t authority_len = path != NULL ? (size_t)(path - authority) :
                          strlen(authority);
    if (authority_len == 0 || authority_len >= sizeof(upstream->host_header))
        return 400;

    char authority_copy[272];
    memcpy(authority_copy, authority, authority_len);
    authority_copy[authority_len] = '\0';
    if (strchr(authority_copy, '@') != NULL) return 400;

    const char *host_start = authority_copy;
    const char *host_end = NULL;
    const char *port = "80";
    if (authority_copy[0] == '[') {
        host_start++;
        host_end = strchr(host_start, ']');
        if (host_end == NULL) return 400;
        if (host_end[1] == ':') port = host_end + 2;
        else if (host_end[1] != '\0') return 400;
    } else {
        const char *colon = strchr(authority_copy, ':');
        host_end = colon != NULL ? colon : authority_copy + authority_len;
        if (colon != NULL) port = colon + 1;
    }
    size_t host_len = (size_t)(host_end - host_start);
    if (host_len == 0 || host_len >= sizeof(upstream->host) ||
        strlen(port) >= sizeof(upstream->port)) return 400;
    for (const char *p = port; *p; p++) {
        if (!isdigit((unsigned char)*p)) return 400;
    }
    char *port_end = NULL;
    errno = 0;
    long port_number = strtol(port, &port_end, 10);
    if (errno == ERANGE || port_end == port || *port_end != '\0' ||
        port_number < 1 || port_number > 65535) return 400;

    memcpy(upstream->host, host_start, host_len);
    upstream->host[host_len] = '\0';
    memcpy(upstream->port, port, strlen(port) + 1);
    memcpy(upstream->host_header, authority_copy, authority_len + 1);

    if (path == NULL || *path == '#') {
        if (origin_capacity < 2) return 400;
        memcpy(origin, "/", 2);
    } else if (*path == '?') {
        int n = snprintf(origin, origin_capacity, "/%s", path);
        if (n < 0 || (size_t)n >= origin_capacity) return 414;
    } else if (*path == '#') {
        if (origin_capacity < 2) return 414;
        memcpy(origin, "/", 2);
    } else {
        const char *fragment = strchr(path, '#');
        size_t path_len = fragment != NULL ? (size_t)(fragment - path) : strlen(path);
        if (path_len == 0 || path_len >= origin_capacity) return 414;
        memcpy(origin, path, path_len);
        origin[path_len] = '\0';
    }
    return 0;
}

int resp_format_head(char *header, size_t capacity, int status,
                     const char *content_type, long content_length,
                     int keep_alive) {
    char date[64] = "Thu, 01 Jan 1970 00:00:00 GMT";
    time_t now = time(NULL);
    struct tm utc;
    if (now != (time_t)-1 && gmtime_r(&now, &utc) != NULL) {
        strftime(date, sizeof(date), "%a, %d %b %Y %H:%M:%S GMT", &utc);
    }

    int length = snprintf(header, capacity,
                          "HTTP/1.1 %d %s\r\n"
                          "Content-Type: %s\r\n"
                          "Content-Length: %ld\r\n"
                          "Connection: %s\r\n"
                          "Date: %s\r\n"
                          "Server: c-webserver-proxy/1.0\r\n\r\n",
                          status, resp_reason(status),
                          content_type != NULL ? content_type : "application/octet-stream",
                          content_length, keep_alive ? "keep-alive" : "close", date);
    if (length < 0 || (size_t)length >= capacity) return -1;
    return length;
}

int resp_send_head(int fd, int status, const char *content_type,
                   long content_length, int keep_alive) {
    char header[1024];
    int length = resp_format_head(header, sizeof(header), status,
                                  content_type, content_length, keep_alive);
    if (length < 0) {
        static const char fallback[] =
            "HTTP/1.1 500 Internal Server Error\r\nContent-Length: 0\r\nConnection: close\r\n\r\n";
        return net_send_all(fd, fallback, sizeof(fallback) - 1);
    }
    return net_send_all(fd, header, (size_t)length);
}

static int valid_port(const char *port) {

// host:port 또는 [IPv6]:port를 고정 크기 필드에 나눠 담습니다.
static int parse_backend(const char *text, struct backend *backend) {
    const char *host_start = text;
    const char *host_end = NULL;
    const char *port_start = NULL;

    if (text[0] == '[') {
        host_start++;
        host_end = strchr(host_start, ']');
        if (host_end == NULL || host_end[1] != ':') return -1;
        port_start = host_end + 2;
    } else {
        const char *colon = strrchr(text, ':');
        if (colon == NULL || colon == text) return -1;
        host_end = colon;
        port_start = colon + 1;
    }

    size_t host_len = (size_t)(host_end - host_start);
    size_t port_len = strlen(port_start);
    if (host_len == 0 || host_len >= sizeof(backend->host) ||
        port_len == 0 || port_len >= sizeof(backend->port) ||
        !valid_port(port_start)) {
        return -1;
    }

    memcpy(backend->host, host_start, host_len);
    backend->host[host_len] = '\0';
    memcpy(backend->port, port_start, port_len + 1);
    return 0;
}

static void destroy_loaded(struct route *routes, int count) {

        if (routes[i].mutex_ready) {
            pthread_mutex_destroy(&routes[i].mutex);
            routes[i].mutex_ready = 0;

// 각 줄을 호스트, 경로 접두사, 백엔드 목록으로 해석합니다.
int route_load(const char *path, struct route *routes, int max) {
    if (path == NULL || routes == NULL || max <= 0) {
        errno = EINVAL;
        return -1;
    }

    FILE *file = fopen(path, "r");
    if (file == NULL) return -1;

    char line[ROUTE_LINE_CAP];
    int count = 0;
    int failed = 0;
    while (fgets(line, sizeof(line), file) != NULL) {
        size_t length = strlen(line);
        if (length > 0 && line[length - 1] != '\n' && !feof(file)) {
            failed = 1;
            errno = EOVERFLOW;
            break;
        }
        if (length > 0 && line[length - 1] == '\n') line[--length] = '\0';
        if (length > 0 && line[length - 1] == '\r') line[--length] = '\0';

        char *cursor = line;
        while (isspace((unsigned char)*cursor)) cursor++;
        if (*cursor == '\0' || *cursor == '#') continue;
        if (count >= max) {
            failed = 1;
            errno = EOVERFLOW;
            break;
        }

        struct route *route = &routes[count];
        memset(route, 0, sizeof(*route));
        char *save = NULL;
        char *host = strtok_r(cursor, " \t", &save);
        char *prefix = strtok_r(NULL, " \t", &save);
        if (host == NULL || prefix == NULL || prefix[0] != '/' ||
            strlen(host) >= sizeof(route->host) ||
            strlen(prefix) >= sizeof(route->prefix)) {
            failed = 1;
            errno = EINVAL;
            break;
        }
        strcpy(route->host, host);
        strcpy(route->prefix, prefix);

        char *backend_text;
        while ((backend_text = strtok_r(NULL, " \t", &save)) != NULL) {
            if (route->nbackends >= ROUTE_BACKEND_MAX ||
                parse_backend(backend_text,
                              &route->backends[route->nbackends]) < 0) {
                failed = 1;
                errno = EINVAL;
                break;
            }
            route->nbackends++;
        }
        if (failed) break;
        if (route->nbackends == 0) {
            failed = 1;
            errno = EINVAL;
            break;
        }

        int result = pthread_mutex_init(&route->mutex, NULL);
        if (result != 0) {
            failed = 1;
            errno = result;
            break;
        }
        route->mutex_ready = 1;
        count++;
    }

    if (ferror(file)) failed = 1;
    fclose(file);
    if (failed) {
        destroy_loaded(routes, count);
        return -1;
    }
    return count;
}

// Host 헤더의 포트 부분을 제외해 규칙 파일의 호스트와 비교합니다.
static int host_matches(const char *rule_host, const char *request_host) {
    if (strcmp(rule_host, "*") == 0) return 1;
    if (request_host == NULL) return 0;

    char host[256];
    size_t length = strlen(request_host);
    if (request_host[0] == '[') {
        const char *closing = strchr(request_host, ']');
        if (closing == NULL) return 0;
        length = (size_t)(closing - request_host - 1);
        request_host++;
    } else {
        const char *colon = strchr(request_host, ':');
        if (colon != NULL) length = (size_t)(colon - request_host);
    }
    if (length == 0 || length >= sizeof(host)) return 0;
    memcpy(host, request_host, length);
    host[length] = '\0';
    return strcasecmp(rule_host, host) == 0;
}

// 가장 긴 경로 접두사를 우선하고, 길이가 같으면 호스트 일치 규칙을 우선합니다.
struct route *route_match(struct route *routes, int count,
                          const char *host, const char *target) {
    if (routes == NULL || count <= 0 || target == NULL) return NULL;

    struct route *best = NULL;
    size_t best_prefix = 0;
    int best_exact_host = 0;
    for (int i = 0; i < count; i++) {
        struct route *candidate = &routes[i];
        size_t prefix_len = strlen(candidate->prefix);
        if (!host_matches(candidate->host, host) ||
            strncmp(target, candidate->prefix, prefix_len) != 0) {
            continue;
        }

        int exact_host = strcmp(candidate->host, "*") != 0;
        if (best == NULL || prefix_len > best_prefix ||
            (prefix_len == best_prefix && exact_host > best_exact_host)) {
            best = candidate;
            best_prefix = prefix_len;
            best_exact_host = exact_host;
        }
    }
    return best;
}

// mutex 안에서 번호를 하나 증가시키고 그 번호에 해당하는 백엔드를 돌려줍니다.
const struct backend *route_pick(struct route *route) {
    if (route == NULL || route->nbackends <= 0) return NULL;
    if (pthread_mutex_lock(&route->mutex) != 0) return NULL;
    unsigned index = route->next % (unsigned)route->nbackends;
    route->next++;
    pthread_mutex_unlock(&route->mutex);
    return &route->backends[index];
}

// 각 규칙의 라운드 로빈 mutex를 제거합니다.
void route_destroy(struct route *routes, int count) {
    if (routes == NULL || count <= 0) return;
    destroy_loaded(routes, count);
}

static int decode_target(const char *target, char *path, size_t capacity) {
    const char *query = strchr(target, '?');
    size_t input_len = query != NULL ? (size_t)(query - target) : strlen(target);
    if (input_len == 0 || target[0] != '/') return 400;

    size_t out = 0;
    for (size_t i = 0; i < input_len; i++) {
        unsigned char value = (unsigned char)target[i];
        if (target[i] == '%') {
            if (i + 2 >= input_len) return 400;
            int hi = hex_value(target[i + 1]);
            int lo = hex_value(target[i + 2]);
            if (hi < 0 || lo < 0) return 400;
            value = (unsigned char)((hi << 4) | lo);
            i += 2;
            if (value == 0) return 400;
        }
        if (out + 1 >= capacity) return 414;
        path[out++] = (char)value;
    }
    path[out] = '\0';

    // 인코딩을 푼 다음 각 경로 조각이 정확히 ".."인지 검사합니다.
    const char *part = path;
    while (*part != '\0') {
        while (*part == '/') part++;
        const char *end = strchr(part, '/');
        size_t length = end != NULL ? (size_t)(end - part) : strlen(part);
        if (length == 2 && part[0] == '.' && part[1] == '.') return 403;
        if (end == NULL) break;
        part = end + 1;
    }
    return 0;
}

    if (decode_result != 0) return send_error_result(fd, decode_result);

        if (path_len + sizeof(index_name) > sizeof(decoded)) {
            return send_error_result(fd, 414);

        memcpy(decoded + path_len, index_name, sizeof(index_name));

        return send_error_result(fd, 500);

        return send_error_result(fd, 414);
    }

    if (realpath(candidate, file_path) == NULL) {
        if (errno == ENOENT) return send_error_result(fd, 404);
        if (errno == EACCES) return send_error_result(fd, 403);
        return send_error_result(fd, 500);
    }

        close(file_fd);
        return -1;
    }

        if (send_result < 0) {
            result = -1;
            break;
        }
    }
    close(file_fd);
    return result;
}

static void test_long_target_and_too_many_headers(void) {
    struct http_request req;
    char *long_target = malloc(2100);
    assert(long_target != NULL);
    const char *prefix = "GET /";
    size_t pos = strlen(prefix);
    memcpy(long_target, prefix, pos);
    memset(long_target + pos, 'a', 2048);
    pos += 2048;
    const char *suffix = " HTTP/1.1\r\nHost: x\r\n\r\n";
    size_t suffix_len = strlen(suffix);
    memcpy(long_target + pos, suffix, suffix_len);
    pos += suffix_len;
    assert(http_parse(long_target, pos, &req) == HTTP_URI_TOO_LONG);
    free(long_target);

    char headers[1200] = "GET / HTTP/1.1\r\nHost: x\r\n";
    for (int i = 0; i < 32; i++) {
        strcat(headers, "X: y\r\n");
    }
    strcat(headers, "\r\n");
    assert(parse(headers, &req) == HTTP_HEADER_TOO_LARGE);
}

static void test_bad_header_and_content_length(void) {
    struct http_request req;
    assert(parse("GET / HTTP/1.1\r\nHost x\r\n\r\n", &req) == HTTP_BAD_REQUEST);
    assert(parse("POST / HTTP/1.1\r\nHost: x\r\nContent-Length: abc\r\n\r\n", &req) == HTTP_BAD_REQUEST);
    assert(parse("POST / HTTP/1.1\r\nHost: x\r\nContent-Length: -1\r\n\r\n", &req) == HTTP_BAD_REQUEST);
    assert(parse("POST / HTTP/1.1\r\nHost: x\r\nContent-Length: 2000000\r\n\r\n", &req) == HTTP_PAYLOAD_TOO_LARGE);
}
