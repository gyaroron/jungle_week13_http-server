static volatile sig_atomic_t g_stop = 0;
static const struct config *g_worker_config = NULL;

// 일꾼이 받은 연결 번호를 기존 연결 처리기에 전달합니다.
static void serve_connection(int fd) {
    struct sockaddr_in peer;
    socklen_t peer_len = sizeof(peer);
    char client_ip[INET_ADDRSTRLEN] = "unknown";
    if (getpeername(fd, (struct sockaddr *)&peer, &peer_len) == 0) {
        if (inet_ntop(AF_INET, &peer.sin_addr, client_ip,
                      sizeof(client_ip)) == NULL) {
            client_ip[0] = '\0';
            memcpy(client_ip, "unknown", sizeof("unknown"));
        }
    }
    conn_handle(fd, client_ip, g_worker_config);
}

// 부분적으로 읽은 라우팅 규칙도 초기화된 mutex만 정리합니다.
static void cleanup_routes(struct route *routes, int route_count) {
    route_destroy(routes, route_count);
    free(routes);
}

static void request_stop(int signal_number) {
    (void)signal_number;
    g_stop = 1;
}

int main(int argc, char **argv) {
    struct config cfg;
    int config_result = config_parse(argc, argv, &cfg);
    if (config_result != 0) {
        return config_result > 0 ? 0 : 2;
    }

    struct route *routes = NULL;
    if (cfg.routes_path != NULL) {
        routes = calloc(ROUTE_MAX, sizeof(*routes));
        if (routes == NULL) {
            server_log_errno("allocate routes", errno);
            return 1;
        }
        int count = route_load(cfg.routes_path, routes, ROUTE_MAX);
        if (count < 0) {
            server_log_errno("load routes", errno);
            cleanup_routes(routes, 0);
            return 1;
        }
        cfg.routes = routes;
        cfg.route_count = count;
        server_log("loaded %d route rules from %s\n", count, cfg.routes_path);
    }

    // 종료 신호를 받으면 accept 반복을 끝내도록 상태만 표시합니다.
    struct sigaction stop_action = {0};
    stop_action.sa_handler = request_stop;
    sigemptyset(&stop_action.sa_mask);
    if (sigaction(SIGINT, &stop_action, NULL) < 0 ||
        sigaction(SIGTERM, &stop_action, NULL) < 0) {
        server_log_errno("sigaction stop", errno);
        cleanup_routes(routes, cfg.route_count);
        return 1;
    }

    // 끊긴 클라이언트에 send해도 SIGPIPE가 서버 프로세스를 끝내지 않게 합니다.
    struct sigaction ignore_pipe = {0};
    ignore_pipe.sa_handler = SIG_IGN;
    sigemptyset(&ignore_pipe.sa_mask);
    if (sigaction(SIGPIPE, &ignore_pipe, NULL) < 0) {
        server_log_errno("sigaction SIGPIPE", errno);
        cleanup_routes(routes, cfg.route_count);
        return 1;
    }

    // net_listen()에 서버 주소와 포트, 대기 큐 크기를 전달해 듣는 소켓을 준비시킵니다.
    int listen_fd = net_listen(cfg.host, cfg.port, cfg.backlog);
    if (listen_fd < 0) {
        // net_listen()이 실패 이유를 이미 출력했으므로 서버를 종료합니다.
        cleanup_routes(routes, cfg.route_count);
        return 1;
    }
    if (net_set_nonblock(listen_fd) < 0) {
        server_log_errno("fcntl listen nonblock", errno);
        close(listen_fd);
        cleanup_routes(routes, cfg.route_count);
        return 1;
    }

    // 서버가 연결을 기다리는 주소와 듣는 소켓 번호를 출력합니다.
    server_log("listening on %s:%d (fd %d)\n", cfg.host, cfg.port, listen_fd);

    // 연결 처리 함수가 요청 처리 설정을 읽도록 주소를 보관합니다.
    g_worker_config = &cfg;

    // 서버를 계속 실행하며 연결 하나씩 차례대로 받습니다.
    while (!g_stop) {
        int ready = net_wait(listen_fd, POLLIN, 1000);
        if (ready == 0) continue;
        if (ready < 0) {
            if (errno == EINTR) continue;
            server_log_errno("poll", errno);
            continue;
        }

        // accept()가 접속한 클라이언트의 주소 정보를 기록할 공간입니다.
        struct sockaddr_in peer;
        socklen_t peer_len = sizeof(peer);

        // 커널에 listen_fd의 대기 연결 하나를 수락해 새 연결 소켓을 만들도록 요청합니다.
        int conn_fd = accept(listen_fd, (struct sockaddr *)&peer, &peer_len);
        if (conn_fd < 0) {
            if (errno == EINTR || errno == EAGAIN || errno == EWOULDBLOCK) {
                continue;
            }
            // 한 연결을 받지 못해도 서버는 종료하지 않고 다음 연결 요청을 기다립니다.
            server_log_errno("accept", errno);
            continue;
        }

        // macOS는 듣는 소켓의 비차단 설정을 물려주므로 연결 소켓을 차단 모드로 되돌립니다.
        if (net_set_blocking(conn_fd) < 0) {
            server_log_errno("fcntl connection blocking", errno);
            close(conn_fd);
            continue;
        }
        if (net_set_timeouts(conn_fd, cfg.timeout_sec) < 0) {
            server_log_errno("setsockopt timeout", errno);
            close(conn_fd);
            continue;
        }

#ifdef SO_NOSIGPIPE
        // macOS 연결 소켓에 쓰기 중 SIGPIPE를 만들지 않도록 설정합니다.
        int no_sigpipe = 1;
        if (setsockopt(conn_fd, SOL_SOCKET, SO_NOSIGPIPE,
                       &no_sigpipe, sizeof(no_sigpipe)) < 0) {
            server_log_errno("setsockopt SO_NOSIGPIPE", errno);
            close(conn_fd);
            continue;
        }
#endif

        // 클라이언트 주소 구조체에 담긴 IPv4 주소를 사람이 읽는 문자열로 바꿉니다.
        char ip[INET_ADDRSTRLEN];
        if (inet_ntop(AF_INET, &peer.sin_addr, ip, sizeof(ip)) == NULL) {
            // 주소 변환에 실패하면 이 연결 소켓을 닫고 다음 연결을 기다립니다.
            server_log_errno("inet_ntop", errno);
            close(conn_fd);
            continue;
        }

        serve_connection(conn_fd);
    }

    close(listen_fd);
    server_log("server stopped\n");
    cleanup_routes(routes, cfg.route_count);
    return 0;
}

static const struct config *g_worker_config = NULL;
static pthread_mutex_t g_active_mutex = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t g_active_changed = PTHREAD_COND_INITIALIZER;
static size_t g_active_threads = 0;

struct connection_job {
    int fd;
};

// 일꾼이 받은 연결 번호를 기존 연결 처리기에 전달합니다.
static void serve_connection(int fd) {
    struct sockaddr_in peer;
    socklen_t peer_len = sizeof(peer);
    char client_ip[INET_ADDRSTRLEN] = "unknown";
    if (getpeername(fd, (struct sockaddr *)&peer, &peer_len) == 0) {
        if (inet_ntop(AF_INET, &peer.sin_addr, client_ip,
                      sizeof(client_ip)) == NULL) {
            client_ip[0] = '\0';
            memcpy(client_ip, "unknown", sizeof("unknown"));
        }
    }
    conn_handle(fd, client_ip, g_worker_config);
}

// 연결별 스레드는 heap에 받은 연결 번호를 꺼낸 뒤, 기존 연결 처리기를 실행합니다.
static void *serve_connection_thread(void *argument) {
    struct connection_job *job = argument;
    int fd = job->fd;
    free(job);

    serve_connection(fd);

    int result = pthread_mutex_lock(&g_active_mutex);
    if (result != 0) {
        server_log_errno("pthread_mutex_lock active", result);
        _exit(1);
    }
    g_active_threads--;
    result = pthread_cond_broadcast(&g_active_changed);
    int unlock_result = pthread_mutex_unlock(&g_active_mutex);
    if (result != 0 || unlock_result != 0) {
        server_log_errno("pthread connection completion",
                         result != 0 ? result : unlock_result);
        _exit(1);
    }
    return NULL;
}

static void cleanup_routes(struct route *routes, int route_count) {
    route_destroy(routes, route_count);

    (void)signal_number;

    int config_result = config_parse(argc, argv, &cfg);

            return 1;

        if (count < 0) {
            server_log_errno("load routes", errno);
            cleanup_routes(routes, 0);
            return 1;
        }

        cfg.route_count = count;
        server_log("loaded %d route rules from %s\n", count, cfg.routes_path);
    }

    struct sigaction stop_action = {0};
    stop_action.sa_handler = request_stop;
    sigemptyset(&stop_action.sa_mask);
    if (sigaction(SIGINT, &stop_action, NULL) < 0 ||
        sigaction(SIGTERM, &stop_action, NULL) < 0) {
        server_log_errno("sigaction stop", errno);
        cleanup_routes(routes, cfg.route_count);
        return 1;
    }

    struct sigaction ignore_pipe = {0};
    ignore_pipe.sa_handler = SIG_IGN;
    sigemptyset(&ignore_pipe.sa_mask);

        server_log_errno("sigaction SIGPIPE", errno);
        cleanup_routes(routes, cfg.route_count);
        return 1;

        cleanup_routes(routes, cfg.route_count);
        return 1;

        server_log_errno("fcntl listen nonblock", errno);
        close(listen_fd);
        cleanup_routes(routes, cfg.route_count);
        return 1;

    g_worker_config = &cfg;

                continue;

            // 한 연결을 받지 못해도 서버는 종료하지 않고 다음 연결 요청을 기다립니다.
            server_log_errno("accept", errno);

            close(conn_fd);
            continue;

            server_log_errno("setsockopt timeout", errno);
            close(conn_fd);
            continue;

            continue;
        }

        if (inet_ntop(AF_INET, &peer.sin_addr, ip, sizeof(ip)) == NULL) {
            // 주소 변환에 실패하면 이 연결 소켓을 닫고 다음 연결을 기다립니다.
            server_log_errno("inet_ntop", errno);
            close(conn_fd);
            continue;
        }
        // 새 연결 소켓 번호와 접속한 클라이언트 주소를 출력합니다.
        server_log("accepted fd %d from %s:%d\n", conn_fd, ip,
                   ntohs(peer.sin_port));

        // 스레드가 연결 번호를 받기 전까지 보관할 heap 공간을 만듭니다.
        struct connection_job *job = malloc(sizeof(*job));
        if (job == NULL) {
            server_log_errno("allocate connection job", errno);
            close(conn_fd);
            continue;
        }
        job->fd = conn_fd;

        pthread_t thread;
        int thread_error = pthread_mutex_lock(&g_active_mutex);
        if (thread_error != 0) {
            server_log_errno("pthread_mutex_lock active", thread_error);
            free(job);
            close(conn_fd);
            continue;
        }
        g_active_threads++;
        thread_error = pthread_create(&thread, NULL,
                                      serve_connection_thread, job);
        if (thread_error != 0) {
            g_active_threads--;
            pthread_mutex_unlock(&g_active_mutex);
            server_log_errno("pthread_create", thread_error);
            free(job);
            close(conn_fd);
            continue;
        }
        int unlock_error = pthread_mutex_unlock(&g_active_mutex);
        if (unlock_error != 0) {
            server_log_errno("pthread_mutex_unlock active", unlock_error);
            _exit(1);
        }

        // 분리한 스레드는 연결 하나를 끝낸 뒤 join 없이 종료합니다.
        thread_error = pthread_detach(thread);
        if (thread_error != 0) {
            server_log_errno("pthread_detach", thread_error);
            thread_error = pthread_join(thread, NULL);
            if (thread_error != 0) {
                server_log_errno("pthread_join fallback", thread_error);
                _exit(1);
            }
        }
    }

    close(listen_fd);
    // detached 스레드가 설정과 라우트를 다 쓸 때까지 main이 정리하지 않습니다.
    int wait_error = pthread_mutex_lock(&g_active_mutex);
    if (wait_error != 0) {
        server_log_errno("pthread_mutex_lock shutdown", wait_error);
        _exit(1);
    }
    while (g_active_threads > 0) {
        wait_error = pthread_cond_wait(&g_active_changed, &g_active_mutex);
        if (wait_error != 0) {
            pthread_mutex_unlock(&g_active_mutex);
            server_log_errno("pthread_cond_wait shutdown", wait_error);
            _exit(1);
        }
    }
    wait_error = pthread_mutex_unlock(&g_active_mutex);
    if (wait_error != 0) {
        server_log_errno("pthread_mutex_unlock shutdown", wait_error);
        _exit(1);
    }
    pthread_cond_destroy(&g_active_changed);
    pthread_mutex_destroy(&g_active_mutex);
    server_log("server stopped\n");
    cleanup_routes(routes, cfg.route_count);
    return 0;
}

// 일꾼이 받은 연결 번호를 기존 연결 처리기에 전달합니다.
static void serve_connection(int fd) {
    struct sockaddr_in peer;
    socklen_t peer_len = sizeof(peer);
    char client_ip[INET_ADDRSTRLEN] = "unknown";
    if (getpeername(fd, (struct sockaddr *)&peer, &peer_len) == 0) {
        if (inet_ntop(AF_INET, &peer.sin_addr, client_ip,
                      sizeof(client_ip)) == NULL) {
            client_ip[0] = '\0';
            memcpy(client_ip, "unknown", sizeof("unknown"));
        }
    }
    conn_handle(fd, client_ip, g_worker_config);
}

static void cleanup_routes(struct route *routes, int route_count) {
    route_destroy(routes, route_count);
    free(routes);
}

static void request_stop(int signal_number) {
    (void)signal_number;
    g_stop = 1;
}

int main(int argc, char **argv) {
    struct config cfg;
    int config_result = config_parse(argc, argv, &cfg);
    if (config_result != 0) {
        return config_result > 0 ? 0 : 2;
    }

    struct route *routes = NULL;
    if (cfg.routes_path != NULL) {
        routes = calloc(ROUTE_MAX, sizeof(*routes));
        if (routes == NULL) {
            server_log_errno("allocate routes", errno);
            return 1;
        }
        int count = route_load(cfg.routes_path, routes, ROUTE_MAX);
        if (count < 0) {
            server_log_errno("load routes", errno);
            cleanup_routes(routes, 0);
            return 1;
        }
        cfg.routes = routes;
        cfg.route_count = count;
        server_log("loaded %d route rules from %s\n", count, cfg.routes_path);
    }

    struct sigaction stop_action = {0};
    stop_action.sa_handler = request_stop;
    sigemptyset(&stop_action.sa_mask);
    if (sigaction(SIGINT, &stop_action, NULL) < 0 ||
        sigaction(SIGTERM, &stop_action, NULL) < 0) {
        server_log_errno("sigaction stop", errno);
        cleanup_routes(routes, cfg.route_count);
        return 1;

        server_log_errno("sigaction SIGPIPE", errno);
        cleanup_routes(routes, cfg.route_count);
        return 1;

        server_log_errno("pool_create", errno);
        close(listen_fd);
        cleanup_routes(routes, cfg.route_count);
        return 1;

        if (ready == 0) continue;

                continue;
            }

            server_log_errno("accept", errno);

        }

        if (net_set_blocking(conn_fd) < 0) {
            server_log_errno("fcntl connection blocking", errno);
            close(conn_fd);
            continue;
        }
        if (net_set_timeouts(conn_fd, cfg.timeout_sec) < 0) {
            server_log_errno("setsockopt timeout", errno);
            close(conn_fd);
            continue;
        }

#ifdef SO_NOSIGPIPE
        // macOS 연결 소켓에 쓰기 중 SIGPIPE를 만들지 않도록 설정합니다.
        int no_sigpipe = 1;
        if (setsockopt(conn_fd, SOL_SOCKET, SO_NOSIGPIPE,
                       &no_sigpipe, sizeof(no_sigpipe)) < 0) {
            server_log_errno("setsockopt SO_NOSIGPIPE", errno);
            close(conn_fd);
            continue;
        }
#endif

        // 클라이언트 주소 구조체에 담긴 IPv4 주소를 사람이 읽는 문자열로 바꿉니다.
        char ip[INET_ADDRSTRLEN];
        if (inet_ntop(AF_INET, &peer.sin_addr, ip, sizeof(ip)) == NULL) {
            // 주소 변환에 실패하면 이 연결 소켓을 닫고 다음 연결을 기다립니다.
            server_log_errno("inet_ntop", errno);
            close(conn_fd);
            continue;
        }
        // 새 연결 소켓 번호와 접속한 클라이언트 주소를 출력합니다.
        server_log("accepted fd %d from %s:%d\n", conn_fd, ip,
                   ntohs(peer.sin_port));

        // 연결 번호를 원형 대기열에 넣고, 일꾼이 연결 처리를 맡게 합니다.
        if (pool_submit(workers, conn_fd) < 0) {
            server_log_errno("pool_submit", errno);
            close(conn_fd);
        }
    }

    close(listen_fd);
    pool_shutdown(workers);
    server_log("server stopped\n");
    cleanup_routes(routes, cfg.route_count);
    return 0;
}

static int cgi_hex(char ch) {
    if (ch >= '0' && ch <= '9') return ch - '0';
    if (ch >= 'a' && ch <= 'f') return ch - 'a' + 10;
    if (ch >= 'A' && ch <= 'F') return ch - 'A' + 10;
    return -1;
}

static int cgi_decode_path(const char *target, char *path, size_t cap,
                           const char **query_out) {
    const char *query = strchr(target, '?');
    size_t len = query != NULL ? (size_t)(query - target) : strlen(target);
    if (query_out != NULL) *query_out = query != NULL ? query + 1 : "";
    if (len == 0 || target[0] != '/') return 400;
    size_t out = 0;
    for (size_t i = 0; i < len; i++) {
        unsigned char ch = (unsigned char)target[i];
        if (target[i] == '%') {
            if (i + 2 >= len) return 400;
            int hi = cgi_hex(target[i + 1]);
            int lo = cgi_hex(target[i + 2]);
            if (hi < 0 || lo < 0) return 400;
            ch = (unsigned char)((hi << 4) | lo);
            i += 2;
            if (ch == 0) return 400;
        }
        if (out + 1 >= cap) return 414;
        path[out++] = (char)ch;
    }
    path[out] = '\0';
    for (const char *part = path; *part != '\0';) {
        while (*part == '/') part++;
        const char *end = strchr(part, '/');
        size_t part_len = end != NULL ? (size_t)(end - part) : strlen(part);
        if (part_len == 2 && part[0] == '.' && part[1] == '.') return 403;
        if (end == NULL) break;
        part = end + 1;
    }
    return 0;
}

static int path_inside(const char *root, const char *file) {
    size_t n = strlen(root);
    return strncmp(root, file, n) == 0 &&
           (file[n] == '\0' || file[n] == '/');

static int write_all(int fd, const char *buf, size_t len) {
    size_t done = 0;
    while (done < len) {
        ssize_t n = write(fd, buf + done, len - done);
        if (n < 0 && errno == EINTR) continue;
        if (n <= 0) return -1;
        done += (size_t)n;
    }
    return 0;
}

static int find_header_end(const char *buf, size_t len, size_t *end_out) {
    for (size_t i = 0; i + 3 < len; i++) {
        if (memcmp(buf + i, "\r\n\r\n", 4) == 0) {
            *end_out = i + 4;
            return 1;
        }
    }
    return 0;
}

    int child_status = 0;
    while (waitpid(pid, &child_status, 0) < 0) {
        if (errno != EINTR) { free(output); return cgi_error(fd, 500); }
    }

        return cgi_error(fd, 500);

        return cgi_error(fd, 500);

            if (name_len == 12 && strncasecmp(output + line, "Content-Type", 12) == 0) {

                content_type[value_len] = '\0';
            }
        }
        line = end + 2;
    }

    int send_result = resp_send_simple(fd, 200, content_type,
                                       output + body_start, output_len - body_start,
                                       req->keep_alive);
    free(output);
    return send_result < 0 ? -1 : 0;
}

static int parse_number(const char *text, int min, int max, int *out) {
    if (text == NULL || *text == '\0') {
        return -1;
    }
    for (const char *p = text; *p != '\0'; p++) {
        if (*p < '0' || *p > '9') {
            return -1;
        }
    }
    errno = 0;
    char *end = NULL;
    long value = strtol(text, &end, 10);
    if (errno == ERANGE || end == text || *end != '\0' ||
        value < min || value > max || value > INT_MAX) {
        return -1;
    }
    *out = (int)value;
    return 0;
}

static void conn_set_state(struct conn *c, enum conn_state next) {
    server_log("fd %d: %s -> %s\n", c->fd,
               conn_state_name(c->state), conn_state_name(next));
    c->state = next;
}

// 요청 버퍼 안에서 CRLF 두 쌍을 길이 제한 안에 찾아냅니다.
static int conn_has_header_end(const char *buffer, size_t length) {
    if (length < 4) return 0;
    for (size_t i = 0; i <= length - 4; i++) {
        if (memcmp(buffer + i, "\r\n\r\n", 4) == 0) return 1;
    }
    return 0;
}

        case HTTP_BAD_REQUEST: return 400;
        case HTTP_URI_TOO_LONG: return 414;
        case HTTP_HEADER_TOO_LARGE: return 431;
        case HTTP_NOT_IMPLEMENTED: return 501;
        case HTTP_VERSION_NOT_SUPPORTED: return 505;
        case HTTP_PAYLOAD_TOO_LARGE: return 413;
        case HTTP_OK:
        case HTTP_INCOMPLETE:
            return 500;
    }
    return 500;
}

void conn_handle(int fd, const char *client_ip, const struct config *cfg) {
    struct conn c = {
        .fd = fd,
        .state = CONN_READING,
        .rlen = 0
    };

    // 현재 단계가 닫힘에 도달할 때까지 이 연결의 처리 단계를 진행합니다.
    while (c.state != CONN_CLOSED) {
        switch (c.state) {
            case CONN_READING: {
                // 앞 요청을 처리하고도 다음 요청 바이트가 남았는지 먼저 확인합니다.
                if (conn_has_header_end(c.rbuf, c.rlen)) {
                    server_log("fd %d: request data available (%zu bytes)\n",
                               c.fd, c.rlen);
                    conn_set_state(&c, CONN_PARSING);
                    break;
                }

                // 아직 비어 있는 버퍼 영역의 크기를 구해 넘칠 공간이 없도록 합니다.
                size_t available = sizeof(c.rbuf) - c.rlen - 1;
                if (available == 0) {
                    server_log("fd %d: request headers exceed buffer capacity\n",
                               c.fd);
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }

                // 커널에 새 요청 바이트를 기존 데이터 뒤에 이어서 rbuf에 복사해 달라고 요청합니다.
                ssize_t n = recv(c.fd, c.rbuf + c.rlen, available, 0);
                if (n < 0) {
                    if (errno == EINTR) break;
                    if (errno == EAGAIN || errno == EWOULDBLOCK) {
                        if (c.rlen > 0) {
                            resp_send_error(c.fd, 408);
                        }
                        server_log("fd %d: request read timed out\n", c.fd);
                    } else {
                        server_log_errno("recv", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                } else if (n == 0) {
                    server_log("fd %d: peer closed\n", c.fd);
                    conn_set_state(&c, CONN_CLOSING);
                } else {
                    // 이번에 읽은 바이트 수를 누적하고 문자열 끝을 표시합니다.
                    c.rlen += (size_t)n;
                    c.rbuf[c.rlen] = '\0';

                    // HTTP 헤더 끝의 빈 줄이 도착했을 때만 요청 읽기를 마칩니다.
                    if (conn_has_header_end(c.rbuf, c.rlen)) {
                        server_log("fd %d: request data available (%zu bytes)\n",
                                   c.fd, c.rlen);
                        conn_set_state(&c, CONN_PARSING);
                    } else if (c.rlen == sizeof(c.rbuf) - 1) {
                        // 용량을 다 썼는데 끝 표시가 없으면 버퍼를 더 쓰지 않고 연결을 닫습니다.
                        server_log("fd %d: request headers exceed buffer capacity\n",
                                   c.fd);
                        conn_set_state(&c, CONN_CLOSING);
                    }
                }
                break;
            }

            case CONN_PARSING: {
                // 원문 요청 줄을 검증하고 method, target, version 필드로 나눕니다.
                struct http_request req;
                enum http_result result = http_parse(c.rbuf, c.rlen, &req);
                server_log("fd %d: parse result %s\n", c.fd,
                           http_result_name(result));
                if (result == HTTP_OK) {
                    c.request_len = req.header_len + (size_t)req.content_length;
                    const char *connection = http_header_get(&req, "Connection");
                    c.keep_alive = strcmp(req.version, "HTTP/1.1") == 0;
                    if (connection != NULL) {
                        if (strcasecmp(connection, "close") == 0) {
                            c.keep_alive = 0;
                        } else if (strcasecmp(connection, "keep-alive") == 0) {
                            c.keep_alive = 1;
                        }
                    }
                    req.keep_alive = c.keep_alive;
                    server_log("fd %d: request %s %s %s; headers=%d bytes=%zu\n",
                               c.fd, req.method, req.target, req.version,
                               req.nheaders, req.header_len);
                    if (req.content_length > 0) {
                        size_t expected = (size_t)req.content_length;
                        c.body = malloc(expected + 1);
                        if (c.body == NULL) {
                            server_log_errno("malloc request body", errno);
                            conn_set_state(&c, CONN_CLOSING);
                            break;
                        }

                        // 헤더와 함께 이미 받은 본문 바이트를 새 저장 공간으로 옮깁니다.
                        size_t received = c.rlen - req.header_len;
                        if (received > expected) {
                            received = expected;
                        }
                        memcpy(c.body, c.rbuf + req.header_len, received);

                        // Content-Length가 가리키는 나머지 바이트만 더 받습니다.
                        while (received < expected) {
                            ssize_t n = recv(c.fd, c.body + received,
                                             expected - received, 0);
                            if (n < 0) {
                                if (errno == EINTR) continue;
                                if (errno == EAGAIN || errno == EWOULDBLOCK) {
                                    resp_send_error(c.fd, 408);
                                } else {
                                    server_log_errno("recv body", errno);
                                }
                                free(c.body);
                                c.body = NULL;
                                conn_set_state(&c, CONN_CLOSING);
                                break;
                            }
                            if (n == 0) {
                                server_log("fd %d: peer closed during request body\n",
                                           c.fd);
                                free(c.body);
                                c.body = NULL;
                                conn_set_state(&c, CONN_CLOSING);
                                break;
                            }
                            received += (size_t)n;
                        }
                        if (c.state == CONN_CLOSING) {
                            break;
                        }
                        c.body[received] = '\0';
                        c.body_len = received;
                        server_log("fd %d: request body bytes=%zu\n",
                                   c.fd, received);
                    }
                    conn_set_state(&c, CONN_ROUTING);
                } else {
                    if (resp_send_error(c.fd, conn_error_status(result)) < 0) {
                        server_log_errno("send error response", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                }
                break;
            }

            case CONN_ROUTING: {
                // 프록시 규칙이 먼저 맞는지 보고, 아니면 정적 파일이나 CGI로 보냅니다.
                struct http_request req;
                enum http_result parse_result = http_parse(c.rbuf, c.rlen, &req);
                if (parse_result != HTTP_OK) {
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                req.keep_alive = c.keep_alive;
                int serve_result = 0;
                int proxy_handled = 0;
                if (cfg->forward_proxy &&
                    (strncmp(req.target, "http://", 7) == 0 ||
                     strncmp(req.target, "https://", 8) == 0)) {
                    serve_result = proxy_forward(c.fd, &req, c.body,
                                                 c.body_len, cfg->timeout_sec,
                                                 client_ip);
                    proxy_handled = 1;
                } else {
                    int reverse_result = proxy_reverse(c.fd, &req, c.body,
                                                       c.body_len, cfg,
                                                       client_ip);
                    if (reverse_result == PROXY_NO_ROUTE) {
                        if (strncmp(req.target, "/cgi-bin/", 9) == 0) {
                            serve_result = cgi_serve(c.fd, cfg, &req,
                                                     c.body, c.body_len);
                        } else {
                            serve_result = static_serve(c.fd, cfg, &req);
                        }
                    } else {
                        serve_result = reverse_result == PROXY_FAILED ? -1 : 0;
                        proxy_handled = 1;
                    }
                }
                if (serve_result < 0) {
                    if (errno != EPIPE && errno != ECONNRESET) {
                        server_log_errno("static_serve", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                c.response_sent = 1;
                if (serve_result > 0 || proxy_handled) c.keep_alive = 0;
                conn_set_state(&c, CONN_RESPONDING);
                break;
            }

            case CONN_RESPONDING: {
                if (c.response_sent) {
                    conn_set_state(&c, CONN_KEEPALIVE_CHECK);
                    break;
                }
                static const char body[] = "hello\n";
                if (resp_send_simple(c.fd, 200, "text/plain; charset=utf-8",
                                     body, sizeof(body) - 1, c.keep_alive) < 0) {
                    if (errno != EPIPE && errno != ECONNRESET) {
                        server_log_errno("send", errno);
                    }
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }
                conn_set_state(&c, CONN_KEEPALIVE_CHECK);
                break;
            }

            case CONN_KEEPALIVE_CHECK:
                if (!c.keep_alive) {
                    conn_set_state(&c, CONN_CLOSING);
                    break;
                }

                // 방금 처리한 요청만 버리고, 다음 요청 바이트는 버퍼 앞으로 당깁니다.
                if (c.request_len > c.rlen) {
                    c.request_len = c.rlen;
                }
                c.rlen -= c.request_len;
                memmove(c.rbuf, c.rbuf + c.request_len, c.rlen);
                c.rbuf[c.rlen] = '\0';
                free(c.body);
                c.body = NULL;
                c.body_len = 0;
                c.response_sent = 0;
                conn_set_state(&c, CONN_READING);
                break;

            case CONN_CLOSING:
                // 이 연결의 처리를 마치고 서버 프로세스의 fd를 커널에 반납합니다.
                free(c.body);
                c.body = NULL;
                close(c.fd);
                conn_set_state(&c, CONN_CLOSED);
                break;

            case CONN_CLOSED:
                break;
        }
    }
}

enum http_result http_parse(const char *buf, size_t len, struct http_request *req) {
    if (buf == NULL || req == NULL) {
        return HTTP_BAD_REQUEST;
    }
    if (!has_header_end(buf, len)) {
        return HTTP_INCOMPLETE;
    }

    // 요청 줄의 끝을 찾습니다. 입력 버퍼는 수정하지 않습니다.
    size_t line_end = 0;
    while (line_end + 1 < len &&
           !(buf[line_end] == '\r' && buf[line_end + 1] == '\n')) {
        line_end++;
    }
    if (line_end + 1 >= len) {
        return HTTP_BAD_REQUEST;
    }

    // 요청 줄에는 method, target, version 사이의 공백이 정확히 두 개 있어야 합니다.
    size_t first_space = line_end;
    size_t second_space = line_end;
    for (size_t i = 0; i < line_end; i++) {
        if (buf[i] == ' ') {
            if (first_space == line_end) {
                first_space = i;
            } else if (second_space == line_end) {
                second_space = i;
            } else {
                return HTTP_BAD_REQUEST;
            }
        }
    }
    if (first_space == 0 || second_space == line_end ||
        second_space == first_space + 1 || second_space + 1 == line_end) {
        return HTTP_BAD_REQUEST;
    }

    size_t method_len = first_space;
    size_t target_start = first_space + 1;

    size_t version_len = line_end - version_start;

    if (target_len >= sizeof(req->target)) {
        return HTTP_URI_TOO_LONG;

          (method_len == 4 && memcmp(buf, "POST", 4) == 0))) {
        return HTTP_NOT_IMPLEMENTED;

    if (!((version_len == 8 && memcmp(buf + version_start, "HTTP/1.1", 8) == 0) ||
          (version_len == 8 && memcmp(buf + version_start, "HTTP/1.0", 8) == 0))) {
        return HTTP_VERSION_NOT_SUPPORTED;
    }

    memset(req, 0, sizeof(*req));
    memcpy(req->method, buf, method_len);
    req->method[method_len] = '\0';
    memcpy(req->target, buf + target_start, target_len);
    req->target[target_len] = '\0';
    memcpy(req->version, buf + version_start, version_len);
    req->version[version_len] = '\0';

            return HTTP_HEADER_TOO_LARGE;

        while (value_start < value_end &&

void server_log(const char *format, ...) {

    va_list args;
    va_start(args, format);
    int length = vsnprintf(line, sizeof(line), format, args);
    va_end(args);

    if (length < 0) return;
    size_t remaining = (size_t)length;
    if (remaining >= sizeof(line)) remaining = sizeof(line) - 1;

    int lock_result = pthread_mutex_lock(&g_log_mutex);
    if (lock_result != 0) return;

    size_t written = 0;
    while (written < remaining) {
        ssize_t count = write(STDERR_FILENO, line + written, remaining - written);
        if (count < 0 && errno == EINTR) continue;
        if (count <= 0) break;
        written += (size_t)count;
    }

    pthread_mutex_unlock(&g_log_mutex);
}

    int listen_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (listen_fd < 0) {
        server_log_errno("socket", errno);
        return -1;
    }

    // 서버를 다시 실행할 때 주소를 재사용할 수 있도록 소켓 옵션을 켭니다.
    int on = 1;
    if (setsockopt(listen_fd, SOL_SOCKET, SO_REUSEADDR, &on, sizeof(on)) < 0) {
        server_log_errno("setsockopt", errno);
        close(listen_fd);
        return -1;
    }

    // bind()가 사용할 IPv4 주소와 포트를 담을 구조체를 준비합니다.
    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_port = htons((uint16_t)port);

    // host 문자열을 구조체가 사용할 IPv4 주소 값으로 바꿉니다.
    int address_result = inet_pton(AF_INET, host, &addr.sin_addr);
    if (address_result != 1) {
        if (address_result == 0) {
            server_log("invalid IPv4 address: %s\n", host);
        } else {
            server_log_errno("inet_pton", errno);
        }
        close(listen_fd);
        return -1;
    }

    // 커널에 listen_fd가 사용할 주소와 포트를 지정해 달라고 요청합니다.
    if (bind(listen_fd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
        server_log_errno("bind", errno);
        close(listen_fd);
        return -1;
    }

    // 커널에 이 소켓을 연결 요청 대기 상태로 바꾸고 대기 큐 크기를 설정하도록 요청합니다.
    if (listen(listen_fd, backlog) < 0) {
        server_log_errno("listen", errno);
        close(listen_fd);
        return -1;
    }

    // 듣는 소켓 번호를 호출한 main()에 돌려줍니다.
    return listen_fd;
}

// 지정한 바이트를 모두 보낼 때까지 send()를 반복합니다.
int net_send_all(int fd, const void *buf, size_t len) {
    const char *bytes = buf;
    size_t sent = 0;

    while (sent < len) {
        ssize_t n = send(fd, bytes + sent, len - sent, 0);
        if (n < 0) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (n == 0) {
            // 전송량이 늘지 않으면 더 진행할 수 없다는 오류로 돌려줍니다.
            errno = EPIPE;
            return -1;
        }
        sent += (size_t)n;
    }

    return 0;
}

int net_wait(int fd, short events, int timeout_ms) {
    struct pollfd pfd = {.fd = fd, .events = events, .revents = 0};
    int result = poll(&pfd, 1, timeout_ms);
    if (result <= 0) return result;
    if (pfd.revents & (events | POLLHUP | POLLERR)) return 1;
    errno = EBADF;
    return -1;
}

int net_connect(const char *host, const char *port, int timeout_ms) {
    if (host == NULL || port == NULL || timeout_ms <= 0) {
        errno = EINVAL;
        return -1;
    }

    struct addrinfo hints;
    memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_STREAM;
    hints.ai_protocol = IPPROTO_TCP;

    struct addrinfo *addresses = NULL;
    int lookup = getaddrinfo(host, port, &hints, &addresses);
    if (lookup != 0) {
        errno = EHOSTUNREACH;
        return -1;
    }

    int saved_error = ECONNREFUSED;
    int connected_fd = -1;
    for (struct addrinfo *address = addresses;
         address != NULL; address = address->ai_next) {
        int fd = socket(address->ai_family, address->ai_socktype,
                        address->ai_protocol);
        if (fd < 0) {
            saved_error = errno;
            continue;
        }
        if (net_set_nonblock(fd) < 0) {
            saved_error = errno;
            close(fd);
            continue;
        }

        int result = connect(fd, address->ai_addr, address->ai_addrlen);
        if (result < 0 && errno != EINPROGRESS && errno != EWOULDBLOCK &&
            errno != EALREADY && errno != EINTR) {
            saved_error = errno;
            close(fd);
            continue;
        }

        if (result < 0) {
            int ready;
            do {
                ready = net_wait(fd, POLLOUT, timeout_ms);
            } while (ready < 0 && errno == EINTR);
            if (ready == 0) {
                saved_error = ETIMEDOUT;
                close(fd);
                continue;
            }
            if (ready < 0) {
                saved_error = errno;
                close(fd);
                continue;
            }

            int socket_error = 0;
            socklen_t error_len = sizeof(socket_error);
            if (getsockopt(fd, SOL_SOCKET, SO_ERROR,
                           &socket_error, &error_len) < 0) {
                saved_error = errno;
                close(fd);
                continue;
            }
            if (socket_error != 0) {
                saved_error = socket_error;
                close(fd);
                continue;
            }
        }

        if (net_set_blocking(fd) < 0) {
            saved_error = errno;
            close(fd);
            continue;
        }

        struct timeval timeout = {
            .tv_sec = timeout_ms / 1000,
            .tv_usec = (timeout_ms % 1000) * 1000
        };
        if (setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO,
                       &timeout, sizeof(timeout)) < 0 ||
            setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO,
                       &timeout, sizeof(timeout)) < 0) {
            saved_error = errno;
            close(fd);
            continue;
        }

#ifdef SO_NOSIGPIPE
        int no_sigpipe = 1;
        if (setsockopt(fd, SOL_SOCKET, SO_NOSIGPIPE,
                       &no_sigpipe, sizeof(no_sigpipe)) < 0) {
            saved_error = errno;
            close(fd);
            continue;
        }
#endif

        connected_fd = fd;
        break;
    }

    freeaddrinfo(addresses);
    if (connected_fd < 0) errno = saved_error;
    return connected_fd;
}

// 일꾼은 큐가 비면 잠들고, 연결 하나를 꺼내 처리한 뒤 다시 기다립니다.
static void *pool_worker(void *argument) {
    struct pool *p = argument;

    for (;;) {
        pthread_mutex_lock(&p->mutex);
        while (p->count == 0 && !p->stopping) {
            pthread_cond_wait(&p->not_empty, &p->mutex);
        }

        if (p->count == 0 && p->stopping) {
            pthread_mutex_unlock(&p->mutex);
            return NULL;
        }

        int fd = p->queue[p->head];
        p->head = (p->head + 1) % p->queue_cap;
        p->count--;
        pthread_cond_signal(&p->not_full);
        pthread_mutex_unlock(&p->mutex);

        p->handler(fd);
    }
}

// mutex·조건 변수·대기열을 준비한 뒤 지정한 수만큼 일꾼을 시작합니다.
struct pool *pool_create(int workers, int queue_cap, void (*handler)(int fd)) {
    if (workers <= 0 || queue_cap <= 0 || handler == NULL) {
        errno = EINVAL;
        return NULL;
    }

    struct pool *p = calloc(1, sizeof(*p));
    if (p == NULL) return NULL;

    p->threads = calloc((size_t)workers, sizeof(*p->threads));
    p->queue = calloc((size_t)queue_cap, sizeof(*p->queue));
    if (p->threads == NULL || p->queue == NULL) goto fail;

    p->worker_count = workers;
    p->queue_cap = (size_t)queue_cap;
    p->handler = handler;

    int result = pthread_mutex_init(&p->mutex, NULL);
    if (result != 0) {
        errno = result;
        goto fail;
    }
    p->mutex_ready = 1;

    result = pthread_cond_init(&p->not_empty, NULL);
    if (result != 0) {
        errno = result;
        goto fail;
    }
    p->not_empty_ready = 1;

    result = pthread_cond_init(&p->not_full, NULL);
    if (result != 0) {
        errno = result;
        goto fail;
    }
    p->not_full_ready = 1;

    for (int i = 0; i < workers; i++) {
        result = pthread_create(&p->threads[i], NULL, pool_worker, p);
        if (result != 0) {
            errno = result;
            goto fail;
        }
        p->started_count++;
    }
    return p;

fail:
    pool_shutdown(p);
    return NULL;
}

// 큐가 가득 찬 동안 기다렸다가 연결 fd를 원형 대기열에 넣습니다.
int pool_submit(struct pool *p, int fd) {
    if (p == NULL) {
        errno = EINVAL;
        return -1;
    }

    int result = pthread_mutex_lock(&p->mutex);
    if (result != 0) {
        errno = result;
        return -1;
    }

    while (p->count == p->queue_cap && !p->stopping) {
        result = pthread_cond_wait(&p->not_full, &p->mutex);
        if (result != 0) {
            pthread_mutex_unlock(&p->mutex);
            errno = result;
            return -1;
        }
    }

    if (p->stopping) {
        pthread_mutex_unlock(&p->mutex);
        errno = ECANCELED;
        return -1;
    }

    p->queue[p->tail] = fd;
    p->tail = (p->tail + 1) % p->queue_cap;
    p->count++;
    pthread_cond_signal(&p->not_empty);
    pthread_mutex_unlock(&p->mutex);
    return 0;
}

// 제출을 막고 대기 중인 일꾼을 깨운 뒤 큐를 비우고 자원을 해제합니다.
void pool_shutdown(struct pool *p) {
    if (p == NULL) return;

    if (p->mutex_ready) {
        pthread_mutex_lock(&p->mutex);
        p->stopping = 1;
        if (p->not_empty_ready) pthread_cond_broadcast(&p->not_empty);
        if (p->not_full_ready) pthread_cond_broadcast(&p->not_full);
        pthread_mutex_unlock(&p->mutex);
    }

    for (int i = 0; i < p->started_count; i++) {
        pthread_join(p->threads[i], NULL);
    }

    if (p->not_full_ready) pthread_cond_destroy(&p->not_full);
    if (p->not_empty_ready) pthread_cond_destroy(&p->not_empty);
    if (p->mutex_ready) pthread_mutex_destroy(&p->mutex);
    free(p->queue);
    free(p->threads);
    free(p);
}

static int append_format(char *buffer, size_t capacity, size_t *used,
                         const char *format, ...) {
    if (*used >= capacity) return -1;
    va_list args;
    va_start(args, format);
    int result = vsnprintf(buffer + *used, capacity - *used, format, args);
    va_end(args);
    if (result < 0 || (size_t)result >= capacity - *used) return -1;
    *used += (size_t)result;
    return 0;
}

static int connection_names_header(const struct http_request *req,
                                   const char *name) {
    for (int i = 0; i < req->nheaders; i++) {
        if (strcasecmp(req->headers[i].name, "Connection") != 0) continue;

        char copy[1024];
        size_t length = strlen(req->headers[i].value);
        if (length >= sizeof(copy)) return 1;
        memcpy(copy, req->headers[i].value, length + 1);

        char *save = NULL;
        for (char *token = strtok_r(copy, ",", &save);
             token != NULL; token = strtok_r(NULL, ",", &save)) {

            while (token_len > 0 &&
                   (token[token_len - 1] == ' ' || token[token_len - 1] == '\t')) {

            if (strcasecmp(token, name) == 0) return 1;
        }
    }
    return 0;

static char *build_request(const struct http_request *req,

                           const char *host_header,
                           size_t body_len, const char *client_ip,
                           size_t *header_len) {

                      request_target) < 0 ||

                      "Host: %s\r\n", host_header) < 0) {

        return NULL;

    for (int i = 0; i < req->nheaders; i++) {
        const struct http_header *header = &req->headers[i];
        if (!valid_header_name(header->name) ||
            !valid_header_value(header->value)) {
            free(buffer);
            errno = EINVAL;
            return NULL;
        }
        if (strcasecmp(header->name, "Content-Length") == 0) {
            had_content_length = 1;
            continue;
        }
        if (strcasecmp(header->name, "Host") == 0 ||
            strcasecmp(header->name, "X-Forwarded-For") == 0 ||
            is_hop_by_hop(req, header->name)) {
            continue;
        }
        if (append_format(buffer, PROXY_HEADER_CAP, &used, "%s: %s\r\n",
                          header->name, header->value) < 0) {
            free(buffer);
            errno = EOVERFLOW;
            return NULL;
        }
    }

    if (had_content_length || body_len > 0) {
        if (append_format(buffer, PROXY_HEADER_CAP, &used,
                          "Content-Length: %zu\r\n", body_len) < 0) {
            free(buffer);
            errno = EOVERFLOW;
            return NULL;
        }
    }
    if (append_format(buffer, PROXY_HEADER_CAP, &used,
                      "X-Forwarded-For: %s\r\nConnection: close\r\n\r\n",
                      client_ip != NULL ? client_ip : "unknown") < 0) {
        free(buffer);
        errno = EOVERFLOW;
        return NULL;
    }

    *header_len = used;
    return buffer;
}

            }
            break;

            break;

                server_log("client response send failed (errno=%d)\n",

            result = PROXY_FAILED;
            break;

static int parse_absolute_target(const char *target, struct upstream *upstream,
                                  char *origin, size_t origin_capacity) {
    if (strncmp(target, "https://", 8) == 0) return 400;
    if (strncmp(target, "http://", 7) != 0) return 400;

    const char *authority = target + 7;
    const char *path = strpbrk(authority, "/?#");
    size_t authority_len = path != NULL ? (size_t)(path - authority) :
                          strlen(authority);
    if (authority_len == 0 || authority_len >= sizeof(upstream->host_header))
        return 400;

    char authority_copy[272];
    memcpy(authority_copy, authority, authority_len);
    authority_copy[authority_len] = '\0';
    if (strchr(authority_copy, '@') != NULL) return 400;

    const char *host_end = NULL;
    const char *port = "80";

        host_start++;
        host_end = strchr(host_start, ']');
        if (host_end == NULL) return 400;
        if (host_end[1] == ':') port = host_end + 2;
        else if (host_end[1] != '\0') return 400;
    } else {
        const char *colon = strchr(authority_copy, ':');
        host_end = colon != NULL ? colon : authority_copy + authority_len;
        if (colon != NULL) port = colon + 1;
    }
    size_t host_len = (size_t)(host_end - host_start);
    if (host_len == 0 || host_len >= sizeof(upstream->host) ||
        strlen(port) >= sizeof(upstream->port)) return 400;
    for (const char *p = port; *p; p++) {
        if (!isdigit((unsigned char)*p)) return 400;
    }
    char *port_end = NULL;
    errno = 0;
    long port_number = strtol(port, &port_end, 10);
    if (errno == ERANGE || port_end == port || *port_end != '\0' ||
        port_number < 1 || port_number > 65535) return 400;

    memcpy(upstream->host, host_start, host_len);
    upstream->host[host_len] = '\0';
    memcpy(upstream->port, port, strlen(port) + 1);
    memcpy(upstream->host_header, authority_copy, authority_len + 1);

    if (path == NULL || *path == '#') {
        if (origin_capacity < 2) return 400;
        memcpy(origin, "/", 2);
    } else if (*path == '?') {
        int n = snprintf(origin, origin_capacity, "/%s", path);
        if (n < 0 || (size_t)n >= origin_capacity) return 414;
    } else if (*path == '#') {
        if (origin_capacity < 2) return 414;
        memcpy(origin, "/", 2);
    } else {
        const char *fragment = strchr(path, '#');
        size_t path_len = fragment != NULL ? (size_t)(fragment - path) : strlen(path);
        if (path_len == 0 || path_len >= origin_capacity) return 414;
        memcpy(origin, path, path_len);
        origin[path_len] = '\0';
    }
    return 0;
}

static enum http_result parse(const char *text, struct http_request *req) {
    return http_parse(text, strlen(text), req);
}

static void test_valid_get_and_trimmed_header(void) {
    const char *text = "GET /a HTTP/1.1\r\nHost: example.test\r\nX-Test:  \thi \t \r\n\r\n";
    struct http_request req;
    assert(parse(text, &req) == HTTP_OK);
    assert(strcmp(req.method, "GET") == 0);
    assert(strcmp(req.target, "/a") == 0);
    assert(strcmp(http_header_get(&req, "x-tEsT"), "hi") == 0);
}

static void test_post_content_length(void) {
    const char *text = "POST /echo HTTP/1.1\r\nHost: example.test\r\nContent-Length: 7\r\n\r\na=1&b=2";
    struct http_request req;
    assert(parse(text, &req) == HTTP_OK);
    assert(req.content_length == 7);
    assert(req.header_len == strlen(text) - 7);
}

static void test_incomplete_and_bad_request_line(void) {
    struct http_request req;
    const char *incomplete = "GET / HTTP/1.1\r\nHost: x\r\n";
    assert(http_parse(incomplete, strlen(incomplete), &req) == HTTP_INCOMPLETE);
    assert(parse("GET /only-one-space\r\n\r\n", &req) == HTTP_BAD_REQUEST);
}

static void test_unsupported_method_and_version(void) {
    struct http_request req;
    assert(parse("PUT / HTTP/1.1\r\nHost: x\r\n\r\n", &req) == HTTP_NOT_IMPLEMENTED);
    assert(parse("GET / HTTP/9.9\r\nHost: x\r\n\r\n", &req) == HTTP_VERSION_NOT_SUPPORTED);
}

static void test_long_target_and_too_many_headers(void) {
    struct http_request req;
    char *long_target = malloc(2100);
    assert(long_target != NULL);
    const char *prefix = "GET /";
    size_t pos = strlen(prefix);
    memcpy(long_target, prefix, pos);
    memset(long_target + pos, 'a', 2048);
    pos += 2048;
    const char *suffix = " HTTP/1.1\r\nHost: x\r\n\r\n";
    size_t suffix_len = strlen(suffix);
    memcpy(long_target + pos, suffix, suffix_len);
    pos += suffix_len;
    assert(http_parse(long_target, pos, &req) == HTTP_URI_TOO_LONG);
    free(long_target);

    char headers[1200] = "GET / HTTP/1.1\r\nHost: x\r\n";
    for (int i = 0; i < 32; i++) {
        strcat(headers, "X: y\r\n");
    }
    strcat(headers, "\r\n");
    assert(parse(headers, &req) == HTTP_HEADER_TOO_LARGE);
}

static void test_bad_header_and_content_length(void) {
    struct http_request req;
    assert(parse("GET / HTTP/1.1\r\nHost x\r\n\r\n", &req) == HTTP_BAD_REQUEST);
    assert(parse("POST / HTTP/1.1\r\nHost: x\r\nContent-Length: abc\r\n\r\n", &req) == HTTP_BAD_REQUEST);
    assert(parse("POST / HTTP/1.1\r\nHost: x\r\nContent-Length: -1\r\n\r\n", &req) == HTTP_BAD_REQUEST);
    assert(parse("POST / HTTP/1.1\r\nHost: x\r\nContent-Length: 2000000\r\n\r\n", &req) == HTTP_PAYLOAD_TOO_LARGE);
}

static void test_host_and_transfer_encoding(void) {

}
